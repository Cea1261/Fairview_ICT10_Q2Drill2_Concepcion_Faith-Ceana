from pyscript import document
from js import alert

members = [
    "Britney Spears",
    "Taylor Swift",
    "Christina Aguilera",
    "Ariana Grande",
    "Bruno Mars",
    "Ed Sheeran",
    "Rosana Swift",
    "Justin Bieber"
]

def update_members():
    text = "ALL MEMBERS:\n\n"
    text += "\n".join(members)
    text += f"\n\nTOTAL MEMBERS: {len(members)}"

    document.getElementById("membersList").textContent = text

def add_member(event):
    first = document.getElementById("firstName").value.strip()
    last = document.getElementById("lastName").value.strip()

    if not first or not last:
        alert("Please enter both names.")
        return

    full_name = f"{first} {last}"
    members.append(full_name)

    update_members()

    alert(f"Added: {full_name}")

    document.getElementById("firstName").value = ""
    document.getElementById("lastName").value = ""

update_members()
